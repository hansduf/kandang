BAB 9 — Dashboard & Ringkasan Data
9.1  DashboardController
// app/Http/Controllers/DashboardController.php

public function index() {
    $stok = StokTelur::first();

    // Produksi hari ini
    $produksiHariIni = ProduksiTelur::whereDate('tanggal_produksi', today())
        ->sum('jumlah_butir');

    // Penjualan bulan ini
    $penjualanBulanIni = Penjualan::whereMonth('tanggal_jual', now()->month)
        ->whereYear('tanggal_jual', now()->year)
        ->sum('total_harga');

    // Jumlah kandang aktif
    $jumlahKandang = Kandang::where('status', 'aktif')->count();

    // Grafik 7 hari terakhir
    $produksi7Hari = ProduksiTelur::whereBetween('tanggal_produksi', [
            now()->subDays(6), now()
        ])
        ->selectRaw('DATE(tanggal_produksi) as tgl, SUM(jumlah_butir) as total')
        ->groupBy('tgl')->orderBy('tgl')->get();

    return view('dashboard.index', compact(
        'stok', 'produksiHariIni', 'penjualanBulanIni', 'jumlahKandang', 'produksi7Hari'
    ));
}
